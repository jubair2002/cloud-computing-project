# ShopHub native run guide: no Docker Compose
# ------------------------------------------------------------
# 1) Repository root
# ------------------------------------------------------------
cd /home/abir/Music/shophub-ecommerce

# ------------------------------------------------------------
# 2) Backend Laravel setup (native PHP process)
# ------------------------------------------------------------
cd back-end

# PHP dependencies
composer install

# Copy backend environment file if needed
cp .env.example .env

# Generate app key
php artisan key:generate

# Set database values in .env
# DB_CONNECTION=mysql
# DB_HOST=127.0.0.1
# DB_PORT=3306
# DB_DATABASE=shophub
# DB_USERNAME=root
# DB_PASSWORD=
# APP_URL=http://127.0.0.1:8000
# FRONTEND_URL=http://127.0.0.1:5173
# SANCTUM_STATEFUL_DOMAINS=127.0.0.1:5173,localhost:5173

# Run database migrations and seed products/categories/users
php artisan migrate
php artisan db:seed

# Optional: make uploaded image files resolvable in storage
php artisan storage:link

# Start the Laravel API in one terminal
php artisan serve --host=0.0.0.0 --port=8000

# Start the queue worker in a second terminal if emails/orders need queue processing
php artisan queue:work --tries=3 --sleep=3 --timeout=60

# ------------------------------------------------------------
# 3) Backend test commands
# ------------------------------------------------------------
cd back-end
php artisan test

# ------------------------------------------------------------
# 4) Frontend Vue + Vite setup (native Node process)
# ------------------------------------------------------------
cd /home/abir/Music/shophub-ecommerce/front-end

# Install Node dependencies
npm install

# Create frontend env if needed
cp .env.example .env

# Set the API URL for the browser build
# VITE_API_BASE_URL=http://127.0.0.1:8000/api

# Build for production or preview locally
npm run build

# Run frontend dev server locallyx`
npm run dev -- --host 0.0.0.0

# Or preview production build locally
npm run preview -- --host 0.0.0.0

# ------------------------------------------------------------
# 5) Frontend test commands
# ------------------------------------------------------------
cd /home/abir/Music/shophub-ecommerce/front-end
npm test
npm run test:watch
npm run test:coverage

# ------------------------------------------------------------
# 6) EC2 deployment notes (same way as local)
# ------------------------------------------------------------
# Do not use Docker Compose for this project.
# Run the same processes directly on the EC2 instance.
#
# Backend EC2:
#   - clone repo
#   - install PHP dependencies with composer install
#   - copy .env.example to .env and set DB_HOST, DB_DATABASE,
#     DB_USERNAME, DB_PASSWORD, APP_URL, FRONTEND_URL
#   - php artisan migrate
#   - php artisan db:seed
#   - php artisan storage:link
#   - php artisan serve --host=0.0.0.0 --port=8000
#   - php artisan queue:work --tries=3 --sleep=3 --timeout=60
#
# Frontend EC2:
#   - build front-end with npm run build
#   - set VITE_API_BASE_URL to the backend EC2 public URL, for example
#     http://<BACKEND_PUBLIC_HOST>:8000/api
#   - host the generated dist folder through Nginx/Apache.
